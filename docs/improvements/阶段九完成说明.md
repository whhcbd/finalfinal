### 孟德尔实验模拟器

#### 核心算法

```typescript
// 配子生成
private getGametes(genotype: string): string[] {
  if (genotype.length === 2) {
    return [genotype[0], genotype[1]]; // Aa → [A, a]
  } else if (genotype.length === 4) {
    // AaBb → [AB, Ab, aB, ab]
    return [
      allele1A + allele2A,
      allele1A + allele2B,
      allele1B + allele2A,
      allele1B + allele2B
    ];
  }
}

// 随机受精模拟
for (let i = 0; i < totalCount; i++) {
  const gamete1 = gametes1[Math.floor(Math.random() * gametes1.length)];
  const gamete2 = gametes2[Math.floor(Math.random() * gametes2.length)];
  const offspring = this.normalizeGenotype(this.combineGametes(gamete1, gamete2));
  counts.set(offspring, (counts.get(offspring) || 0) + 1);
}

// 卡方检验
let chiSquareValue = 0;
counts.forEach((observed, genotype) => {
  const expected = (this.getExpectedPercentage(genotype) / 100) * total;
  chiSquareValue += Math.pow(observed - expected, 2) / expected;
});
```

### 自然选择模拟器

#### 种群初始化

```typescript
private initializePopulation() {
  const freqA = this._initialFreqA;
  const freqAA = freqA * freqA;
  const freqAa = 2 * freqA * (1 - freqA);
  const freqaa = (1 - freqA) * (1 - freqA);

  for (let i = 0; i < this._populationSize; i++) {
    const rand = Math.random();
    let genotype: 'AA' | 'Aa' | 'aa';

    if (rand < freqAA) genotype = 'AA';
    else if (rand < freqAA + freqAa) genotype = 'Aa';
    else genotype = 'aa';

    this.population.push({
      id: i,
      genotype,
      phenotype: this.getPhenotype(genotype),
      x: Math.random() * 100,
      y: Math.random() * 100,
      alive: true
    });
  }
}
```

#### 选择压力算法

```typescript
private getFitness(phenotype: 'dark' | 'medium' | 'light'): number {
  const env = this._environmentType;
  const strength = this._selectionStrength;

  if (env === 'dark') {
    if (phenotype === 'dark') return 1.0;
    if (phenotype === 'medium') return 1.0 - (strength * 0.3);
    return 1.0 - strength;
  } else {
    if (phenotype === 'light') return 1.0;
    if (phenotype === 'medium') return 1.0 - (strength * 0.3);
    return 1.0 - strength;
  }
}

private applySelection() {
  this.population.forEach(individual => {
    if (!individual.alive) return;
    const fitness = this.getFitness(individual.phenotype);
    if (Math.random() > fitness) {
      individual.alive = false;
    }
  });
}
```

#### Canvas 渲染

```typescript
private renderCanvas() {
  const canvas = this.shadowRoot?.getElementById('population-canvas') as HTMLCanvasElement;
  const ctx = canvas.getContext('2d');

  // 绘制背景（深色/浅色环境）
  ctx.fillStyle = this._environmentType === 'dark' ? '#1f2937' : '#f3f4f6';
  ctx.fillRect(0, 0, canvas.width, canvas.height);

  // 绘制个体
  this.population.forEach(individual => {
    if (!individual.alive) return;
    const x = (individual.x / 100) * canvas.width;
    const y = (individual.y / 100) * canvas.height;

    ctx.beginPath();
    ctx.arc(x, y, 4, 0, 2 * Math.PI);
    ctx.fillStyle = individual.phenotype === 'dark' ? '#111827' :
                    individual.phenotype === 'medium' ? '#6b7280' : '#e5e7eb';
    ctx.fill();
  });
}
```
